cmake_minimum_required(VERSION 3.24)
project(tierbench LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(TIERBENCH_CUDA "Build the CUDA plugin" OFF)
option(TIERBENCH_HIP "Build the HIP plugin" OFF)

set(TIERBENCH_PLUGIN_DIR ${CMAKE_BINARY_DIR}/plugins)

add_executable(tierbench src/runner.cpp src/host.cpp)
target_include_directories(tierbench PRIVATE include)
target_link_libraries(tierbench PRIVATE ${CMAKE_DL_LIBS})

function(tierbench_plugin name)
  add_library(${name} MODULE ${ARGN})
  target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/include ${PROJECT_SOURCE_DIR}/plugins)
  set_target_properties(${name} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY ${TIERBENCH_PLUGIN_DIR})
endfunction()

tierbench_plugin(cpu_memory plugins/cpu_memory.cpp)
tierbench_plugin(storage_posix plugins/storage_posix.cpp)

if(APPLE)
  enable_language(OBJCXX)
  tierbench_plugin(metal_memory plugins/metal_memory.mm)
  set_target_properties(metal_memory PROPERTIES OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED ON OBJCXX_EXTENSIONS OFF)
  target_compile_options(metal_memory PRIVATE -fobjc-arc)
  target_link_libraries(metal_memory PRIVATE "-framework Foundation" "-framework Metal")
  tierbench_plugin(metal_prefill plugins/metal_prefill.mm)
  set_target_properties(metal_prefill PROPERTIES OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED ON OBJCXX_EXTENSIONS OFF)
  target_compile_options(metal_prefill PRIVATE -fobjc-arc)
  target_link_libraries(metal_prefill PRIVATE "-framework Foundation" "-framework Metal"
                                              "-framework MetalPerformanceShaders")
endif()

if(TIERBENCH_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  tierbench_plugin(gpu_cuda plugins/gpu_plugin.cpp plugins/gpu_kernels.cu)
  set_target_properties(gpu_cuda PROPERTIES CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Shared)
  target_link_libraries(gpu_cuda PRIVATE CUDA::cudart)
  tierbench_plugin(cuda_prefill plugins/gpu_prefill.cpp)
  target_link_libraries(cuda_prefill PRIVATE CUDA::cudart CUDA::cublas)
  tierbench_plugin(cuda_placement plugins/gpu_placement.cpp plugins/gpu_kernels.cu)
  set_target_properties(cuda_placement PROPERTIES CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Shared)
  target_link_libraries(cuda_placement PRIVATE CUDA::cudart)
endif()

if(TIERBENCH_HIP)
  enable_language(HIP)
  find_package(hip REQUIRED)
  configure_file(plugins/gpu_kernels.cu ${CMAKE_CURRENT_BINARY_DIR}/gpu_kernels.hip COPYONLY)
  tierbench_plugin(gpu_hip plugins/gpu_plugin.cpp ${CMAKE_CURRENT_BINARY_DIR}/gpu_kernels.hip)
  target_compile_definitions(gpu_hip PRIVATE TIERBENCH_HIP)
  target_link_libraries(gpu_hip PRIVATE hip::host)
  tierbench_plugin(hip_placement plugins/gpu_placement.cpp ${CMAKE_CURRENT_BINARY_DIR}/gpu_kernels.hip)
  target_compile_definitions(hip_placement PRIVATE TIERBENCH_HIP)
  target_link_libraries(hip_placement PRIVATE hip::host)
  find_package(hipblas REQUIRED)
  tierbench_plugin(hip_prefill plugins/gpu_prefill.cpp)
  target_compile_definitions(hip_prefill PRIVATE TIERBENCH_HIP)
  target_link_libraries(hip_prefill PRIVATE hip::host roc::hipblas)
endif()
