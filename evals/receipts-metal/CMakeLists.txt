cmake_minimum_required(VERSION 3.20)
project(receipts_metal_capture LANGUAGES C CXX)
if(NOT APPLE)
    message(FATAL_ERROR "Metal receipt capture requires macOS and a Metal device")
endif()
set(NATIVE_SOURCE "" CACHE PATH "Matching prepared native source checkout")
if(NOT EXISTS "${NATIVE_SOURCE}/ggml/CMakeLists.txt")
    message(FATAL_ERROR "NATIVE_SOURCE must name the prepared pinned source")
endif()
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
add_subdirectory("${NATIVE_SOURCE}/ggml" ggml)
add_executable(receipts-metal-capture capture.cpp)
target_compile_features(receipts-metal-capture PRIVATE cxx_std_17)
target_compile_options(receipts-metal-capture PRIVATE -Wall -Wextra -Werror)
target_link_libraries(receipts-metal-capture PRIVATE ggml)
